=== titel
Ein Interface komplett umsetzen
=== erklaerung
Schnittstellen verbinden auch Klassen, die **nichts miteinander zu tun haben**. Ein Schiedsrichter ist keine Figur – bewegen kann er sich trotzdem.

Code, der nur das Interface kennt, funktioniert dann mit beiden:

```java
static void allesVerschieben(Bewegung[] objekte, int dx, int dy) { ... }
```

Jeder Nutzer kann sich darauf verlassen, dass die Methoden des Interfaces vorhanden sind – egal, wie sie umgesetzt sind (Skript S. 40).
=== aufgabe
1. `Schiedsrichter` implementiert **alle fünf** Methoden von `Bewegung` selbst (mit eigenen Variablen `x` und `y`).
2. `allesVerschieben(Bewegung[] objekte, int dx, int dy)` verschiebt jedes Objekt um `dx` und `dy`.

class BewKreis extends KreisF implements Bewegung {

    BewKreis(double r) {
        super(r);
    }

    public void moveX(int mx) {
        xp = xp + mx;
    }

    public void moveY(int my) {
        yp = yp + my;
    }
}

abstract class Figur {
    protected int xp;
    protected int yp;

    public void setPos(int xp, int yp) {
        this.xp = xp;
        this.yp = yp;
    }

    public int getX() {
        return xp;
    }

    public int getY() {
        return yp;
    }

    abstract double flaeche();

    abstract double umfang();

    boolean istGroesserAls(Figur andere) {
        return this.flaeche() > andere.flaeche();
    }
}

class KreisF extends Figur {
    private double r;

    KreisF(double r) {
        this.r = r;
    }

    @Override
    double flaeche() {
        return Math.PI * r * r;
    }

    @Override
    double umfang() {
        return 2 * Math.PI * r;
    }
}

class RechteckF extends Figur {
    private double l;
    private double h;

    RechteckF(double l, double h) {
        this.l = l;
        this.h = h;
    }

    @Override
    double flaeche() {
        return l * h;
    }

    @Override
    double umfang() {
        return 2 * (l + h);
    }
}

=== loesung
public class Main {

    static void allesVerschieben(Bewegung[] objekte, int dx, int dy) {
        for (Bewegung b : objekte) {
            b.moveX(dx);
            b.moveY(dy);
        }
    }

    public static void main(String[] args) {
        Schiedsrichter s = new Schiedsrichter();
        s.setPos(10, 10);
        Bewegung[] alle = {s};
        allesVerschieben(alle, 3, 1);
        System.out.println(s.getX() + "," + s.getY());
    }
}
