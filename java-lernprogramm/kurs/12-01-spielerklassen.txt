=== titel
Projekt 1/3: Die Spielerklassen
=== erklaerung
Im Abschlussprojekt baust du in drei Schritten eine **Kaderverwaltung** für eine Eishockeymannschaft – ein echtes Konsolenprogramm mit Menü. Dabei kommt alles aus dem Skript zusammen.

So hängen die Klassen zusammen:

```
        «abstract» Person                  «interface» Auswertbar
        name, anzahlPersonen (static)       statistik()
        rolle() {abstract}
               △                                  △
               │                                  ┆ implements
        «abstract» Spieler ┄┄┄┄┄┄┄┄┄┄┄┄┄┄┄┄┄┄┄┄┄┄┄┄┄┘
        nummer, toString()
          △              △
     Feldspieler      Torhueter
     tore, vorlagen   paraden, gegentore
```

`Person`, `Auswertbar` und `Spieler` sind schon fertig. Schau sie dir genau an: abstrakte Klasse, Klassenvariable, Interface, `super(...)` und das überschriebene `toString()`, das die abstrakten Methoden `rolle()` und `statistik()` aufruft.
=== aufgabe
Vervollständige `Feldspieler` und `Torhueter`:

**Feldspieler**
- `torErzielen()`, `vorlageGeben()`, `punkte()` (Tore + Vorlagen)
- `rolle()` → `"Feldspieler"`
- `statistik()` → `"Tore: 2  Vorlagen: 1  Punkte: 3"` (je **zwei** Leerzeichen zwischen den Teilen)

**Torhueter**
- `parade()`, `gegentor()`
- `fangquote()` in Prozent: Paraden / (Paraden + Gegentore) · 100, auf **eine** Nachkommastelle gerundet. Ohne Schüsse: `0.0`
- `rolle()` → `"Torhüter"`
- `statistik()` → `"Paraden: 2  Gegentore: 1  Fangquote: 66.7 %"`

abstract class Person {
    private static int anzahlPersonen = 0;
    private final String name;

    Person(String name) {
        this.name = name;
        anzahlPersonen++;
    }

    String getName() {
        return name;
    }

    static int getAnzahlPersonen() {
        return anzahlPersonen;
    }

    abstract String rolle();
}

interface Auswertbar {
    String statistik();
}

abstract class Spieler extends Person implements Auswertbar {
    private final int nummer;

    Spieler(String name, int nummer) {
        super(name);
        this.nummer = nummer;
    }

    int getNummer() {
        return nummer;
    }

    @Override
    public String toString() {
        return "#" + nummer + " " + getName() + " (" + rolle() + ")  " + statistik();
    }
}

class Feldspieler extends Spieler {
    private int tore;
    private int vorlagen;

    Feldspieler(String name, int nummer) {
        super(name, nummer);
    }

    void torErzielen() {
        tore++;
    }

    void vorlageGeben() {
        vorlagen++;
    }

    int punkte() {
        return tore + vorlagen;
    }

    @Override
    String rolle() {
        return "Feldspieler";
    }

    @Override
    public String statistik() {
        return "Tore: " + tore + "  Vorlagen: " + vorlagen + "  Punkte: " + punkte();
    }
}

class Torhueter extends Spieler {
    private int paraden;
    private int gegentore;

    Torhueter(String name, int nummer) {
        super(name, nummer);
    }

    void parade() {
        paraden++;
    }

    void gegentor() {
        gegentore++;
    }

    double fangquote() {
        int schuesse = paraden + gegentore;
        if (schuesse == 0) {
            return 0.0;
        }
        return Math.round(1000.0 * paraden / schuesse) / 10.0;
    }

    @Override
    String rolle() {
        return "Torhüter";
    }

    @Override
    public String statistik() {
        return "Paraden: " + paraden + "  Gegentore: " + gegentore + "  Fangquote: " + fangquote() + " %";
    }
}
